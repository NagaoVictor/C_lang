Example
	x + y Adds together two values
	x - y Subtracts one value from another
	x * y Multiplies two values
	x / y Divides one value by another
	x % y Returns the division remainder
	++x   Increases the value of a variable by 1
	--x   Decreases the value of a variable by 1
	
	
--------------------------------Assignment----------------------------
Assigment example
	=   Iqual
	
	+=  Increment
	
	-=	Decrement
	
	*=  Multiply and assign
	int x = 5;
	x *= 3;   // same as: x = x * 3;
	printf("%d", x); // output: 15

	/=	Divide and assign
	int x = 10;
	x /= 2;   // same as: x = x / 2;
	printf("%d", x); // output: 5

	%=	Modulus and assign
	int x = 10;
	x %= 3;   // same as: x = x % 3;
	printf("%d", x); // output: 1

	&=	Bitwise AND and assign
	int x = 6;   // 6 = 110 (binary)
	x &= 3;      // 3 = 011 → (110 & 011) = 010
	printf("%d", x); // output: 2

	|=  Bitwise OR and assign
	int x = 6;   // 110
	x |= 3;      // 011 → (110 | 011) = 111
	printf("%d", x); // output: 7

	^=  Bitwise XOR and assign
	int x = 6;   // 110
	x ^= 3;      // 011 → (110 ^ 011) = 101
	printf("%d", x); // output: 5

	>>= Shift right and assign
	int x = 8;   // 1000
	x >>= 2;     // move bits 2 places to the right → 0010
	printf("%d", x); // output: 2

	<<= Shift left and assign
	int x = 3;   // 0011
	x <<= 2;     // move bits 2 places to the left → 1100
	printf("%d", x); // output: 12
