type variableName = value;

int myNum = 15;
printf("%d", myNum);  // Outputs 15
----------------------------------------------------------------------
//OverWrite
int myNum = 15;  // myNum is 15 
myNum = 10;  // Now myNum is 10
----------------------------------------------------------------------
//MultOperation
int x = 5, y = 6, z = 50;
printf("%d", x + y + z);

int x, y, z;
x = y = z = 50;
printf("%d", x + y + z);
----------------------------------------------------------------------
// Number Data Types
int studentID = 15;
int studentAge = 23;
float studentFee = 75.25;
char studentGrade = 'B';
----------------------------------------------------------------------
// Char Data Types
char myGrade = 'A';
printf("%c", myGrade);
----------------------------------------------------------------------
//Print Data Types
%d & %i ---> int          2 or 4 bytes
%f & %F ---> float        4 bytes
%lfi    ---> Double       8 bytes
%c      ---> Char         1 bytes
%s      ---> String       Indefied
----------------------------------------------------------------------
//MultChar
char a = 65, b = 66, c = 67;
printf("%c", a);
printf("%c", b);
printf("%c", c);

C Extended Types

%hd      ---> short int  (2 bytes)

%u       ---> unsigned int (2 or 4 bytes)

0 to 4,294,967,295 (4 bytes)

%ld      ---> long int  (4 or 8 bytes)
-2,147,483,648 to 2,147,483,647 (4 bytes)
-9,223,372,036,854,775,808 to 9,223,372,036,854,775,807 (8 bytes)


%lld     ---> long long int (8 bytes)
-9,223,372,036,854,775,808 to 9,223,372,036,854,775,807


%lu      ---> unsigned long int    (4 or 8 bytes)
0 to 4,294,967,295 (4 bytes)
0 to 18,446,744,073,709,551,615 (8 bytes)

%llu     ---> unsigned long long int   (8 bytes)
0 to 18,446,744,073,709,551,615

 
%Lf      ---> long double  (8, 12, or 16 bytes)
