---------------While Loop---------------------

int i = 0;

while (i < 5) {
  printf("%d\n", i);
  i++;
}

--------------Do/While Loop--------------------
int i = 0;

do {
  printf("%d\n", i);
  i++;
}
while (i < 5);
--------------For Loop-------------------------
int i;

for (i = 0; i < 5; i++) {
  printf("%d\n", i);
}
--------------Nested Loops-------------------------
int i, j;

// Outer loop
for (i = 1; i <= 2; ++i) {
  printf("Outer: %d\n", i);  // Executes 2 times

  // Inner loop
  for (j = 1; j <= 3; ++j) {
    printf(" Inner: %d\n", j);  // Executes 6 times (2 * 3)
  }
}

--------------Break-------------------------
int i;

for (i = 0; i < 10; i++) {
  if (i == 4) {
    break;
  }
  printf("%d\n", i);
}
--------------Continue-------------------------
int i;

for (i = 0; i < 10; i++) {
  if (i == 4) {
    continue;
  }
  printf("%d\n", i);
}

--------------Combining Break and Continue-------------------------
int i;

for (i = 0; i < 6; i++) {
  if (i == 2) {
    continue;
  }
  if (i == 4) {
    break;
  }
  printf("%d\n", i);
}
--------------Break and Continue in While Loop-------------------------
int i = 0;

while (i < 10) {
  if (i == 4) {
    break;
  }
  printf("%d\n", i);
  i++;
}
